rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /writeoffs/{doc} {
      allow create: if request.resource.data._auth_key == "placeholder(як і просив)";
      allow read, update, delete: if false;
    }

    match /games41_users/{userId} {
      allow read: if true;
      allow create: if !exists(/databases/$(database)/documents/games41_users/$(userId))
                    && request.resource.data.keys().hasOnly(['passwordHash', 'createdAt']);
      allow update, delete: if false;
    }

    match /games41_scores/{scoreId} {
      allow read: if true;
      allow write: if request.resource.data.keys().hasOnly(['gameName', 'player', 'score', 'ts']);
    }

    match /games41_wordle_users/{userId} {
      allow read: if true;
      allow write: if request.resource.data.keys().hasOnly(['streak', 'lastWinDate', 'updatedAt']);
    }

    match /games41_wordle_results/{resultId} {
      allow read: if true;
      allow create, update: if request.resource.data.keys().hasOnly(['username', 'date', 'lang', 'attempts', 'won', 'shareText', 'ts']);
      allow delete: if true;
    }

    match /games41_quiplash_prompt_history/{promptId} {
      allow read: if true;
      allow write: if request.resource.data.keys().hasOnly(['lastUsedAt']);
    }
  }
}
